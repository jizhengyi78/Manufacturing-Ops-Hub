## 一、PostgreSQL 关系库设计

### 1.1 用户与权限

```sql
-- 用户表
CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username        VARCHAR(50)  NOT NULL UNIQUE,
    display_name    VARCHAR(100) NOT NULL,          -- 真实姓名
    role            VARCHAR(20)  NOT NULL,          -- worker/maintainer/shift_lead/workshop_director/process_engineer/plant_manager
    workshop_id     VARCHAR(50),                    -- 所属车间, plant_manager 为 NULL
    phone           VARCHAR(20),
    is_active       BOOLEAN      DEFAULT TRUE,
    created_at      TIMESTAMPTZ  DEFAULT NOW(),
    updated_at      TIMESTAMPTZ  DEFAULT NOW()
);
CREATE INDEX idx_users_role ON users(role);
CREATE INDEX idx_users_workshop ON users(workshop_id);

-- 角色权限表
CREATE TABLE roles (
    id              VARCHAR(20) PRIMARY KEY,        -- 角色编码
    name            VARCHAR(50) NOT NULL,            -- 角色名称
    permissions     JSONB       NOT NULL DEFAULT '{}', -- 权限位图
    -- 权限示例: {"knowledge:read":true, "diagnosis:use":true, "scheduling:read":false, "cost:read":false}
    created_at      TIMESTAMPTZ DEFAULT NOW()
);

-- 权限审计日志
CREATE TABLE permission_audit (
    id              BIGSERIAL PRIMARY KEY,
    user_id         UUID        NOT NULL,
    action          VARCHAR(100) NOT NULL,           -- 尝试的操作
    resource        VARCHAR(200) NOT NULL,           -- 访问的资源
    result          VARCHAR(20)  NOT NULL,           -- allowed / denied
    request_payload JSONB,                           -- 原始请求载荷
    ip_address      VARCHAR(45),
    created_at      TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_audit_user ON permission_audit(user_id, created_at DESC);
CREATE INDEX idx_audit_denied ON permission_audit(created_at DESC) WHERE result='denied';
```

### 1.2 文档与知识库

```sql
-- 文档元数据表
CREATE TABLE documents (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    doc_id          VARCHAR(100) NOT NULL UNIQUE,    -- 业务唯一ID
    title           VARCHAR(500) NOT NULL,
    doc_type        VARCHAR(50)  NOT NULL,           -- equipment_manual / sop / fault_case / quality_standard / alarm_code
    file_format     VARCHAR(20),                     -- pdf / docx / xlsx
    file_path       TEXT,                            -- 原始文件存储路径
    workshop_id     VARCHAR(50),                     -- 关联车间
    equipment_model VARCHAR(200),                    -- 关联设备型号
    classification  VARCHAR(20)  DEFAULT 'internal', -- public / internal / confidential / secret
    version         VARCHAR(20)  DEFAULT '1.0',
    status          VARCHAR(20)  DEFAULT 'active',   -- active / archived / obsolete
    chunk_count     INT          DEFAULT 0,
    file_size_bytes BIGINT,
    ingested_at     TIMESTAMPTZ,
    updated_at      TIMESTAMPTZ  DEFAULT NOW()
);
CREATE INDEX idx_docs_type ON documents(doc_type);
CREATE INDEX idx_docs_status ON documents(status);
CREATE INDEX idx_docs_workshop ON documents(workshop_id);
CREATE INDEX idx_docs_equipment ON documents(equipment_model);

-- 双写事务记录表 (ES/Milvus同步)
CREATE TABLE ingestion_transactions (
    id              BIGSERIAL PRIMARY KEY,
    doc_id          VARCHAR(100) NOT NULL,
    chunk_id        VARCHAR(100) NOT NULL UNIQUE,
    milvus_ok       BOOLEAN      DEFAULT FALSE,
    es_ok           BOOLEAN      DEFAULT FALSE,
    status          VARCHAR(20)  DEFAULT 'pending',  -- pending / done / anomaly
    retry_count     INT          DEFAULT 0,
    last_error      TEXT,
    created_at      TIMESTAMPTZ  DEFAULT NOW(),
    updated_at      TIMESTAMPTZ  DEFAULT NOW()
);
CREATE INDEX idx_ingestion_status ON ingestion_transactions(status, created_at);
CREATE INDEX idx_ingestion_doc ON ingestion_transactions(doc_id);

-- 离线同步去重表
CREATE TABLE dedup_log (
    id              BIGSERIAL PRIMARY KEY,
    offline_uuid    UUID        NOT NULL,
    online_uuid     UUID,                            -- 匹配到的在线文档UUID
    similarity_score FLOAT,
    action          VARCHAR(20)  NOT NULL,            -- kept_online / kept_offline / merged
    reviewer        VARCHAR(100),
    reviewed_at     TIMESTAMPTZ,
    created_at      TIMESTAMPTZ  DEFAULT NOW()
);
CREATE UNIQUE INDEX idx_dedup_offline ON dedup_log(offline_uuid);
```

### 1.4 知识案例

```sql
-- 知识案例表
CREATE TABLE knowledge_cases (
    id              BIGSERIAL PRIMARY KEY,
    case_id         VARCHAR(100) NOT NULL UNIQUE,

    fault_code      VARCHAR(50),
    fault_category  VARCHAR(100) NOT NULL,            -- 故障大类

    status          VARCHAR(20) DEFAULT 'active',     -- draft / active / archived / obsolete
    review_status   VARCHAR(20),                      -- pending_review / approved / rejected
    reviewer_id     UUID,

    last_accessed   TIMESTAMPTZ,
```

### 1.5 对话与记忆

```sql
-- 对话记录表
CREATE TABLE conversations (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id      VARCHAR(100) NOT NULL,
    user_id         UUID        NOT NULL,
    role            VARCHAR(20)  NOT NULL,            -- user / assistant / system
    content         TEXT        NOT NULL,
    citations       JSONB,                            -- 引用来源 [{"doc_id":"...","title":"...","chunk_id":"..."}]
    token_count     INT,
    model_used      VARCHAR(50),                      -- 使用的LLM模型
    model_cost      DECIMAL(10,6),                    -- 本次调用费用
    latency_ms      INT,                              -- 响应延迟
    feedback        VARCHAR(10),                      -- good / bad / null
    created_at      TIMESTAMPTZ  DEFAULT NOW()
);
CREATE INDEX idx_conv_session ON conversations(session_id, created_at);
CREATE INDEX idx_conv_user ON conversations(user_id, created_at DESC);

-- Token 消耗统计表 (按日/车间/Agent聚合)
CREATE TABLE token_usage_daily (
    id              BIGSERIAL PRIMARY KEY,
    date            DATE        NOT NULL,
    workshop_id     VARCHAR(50) NOT NULL,
    agent           VARCHAR(50) NOT NULL,             -- knowledge / diagnosis / inspection / scheduling / quality / report
    model           VARCHAR(50) NOT NULL,
    prompt_tokens   BIGINT      DEFAULT 0,
    completion_tokens BIGINT    DEFAULT 0,
    cost            DECIMAL(12,6) DEFAULT 0,
    request_count   INT         DEFAULT 0,
    UNIQUE(date, workshop_id, agent, model)
);
CREATE INDEX idx_token_date ON token_usage_daily(date DESC);
```

### 2.3 机密/绝密文档 Collection

| 字段 | 类型 | 说明 |
|------|------|------|
| chunk_id | VARCHAR(100) | 主键 |
| doc_id | VARCHAR(100) | 所属文档ID |

| content | VARCHAR(65535) | 分块文本内容 |

| classification | VARCHAR(20) | confidential / secret |
| access_roles | ARRAY(VARCHAR) | 允许访问的角色列表 |
| **embedding** | FLOAT_VECTOR(1024) | 向量 |
| created_at | INT64 | 时间戳 |

**索引配置**：
- 索引类型: IVF_FLAT
- 度量方式: COSINE
- 分区键: workshop_id
- 鉴权: 检索层校验角色 + 车间双维度

### 2.4 检索时的 Collection 选择逻辑

```
查询请求 (含 user_role, workshop_id, classification_level)
  │
  ├── classification_level = public/internal
  │     → 查 Collection 2.1 (通用文档), partition_key = workshop_id
  │
  ├── classification_level = confidential/secret + user_role in (workshop_director, process_engineer, plant_manager)
  │     → 查 Collection 2.3 (机密文档), partition_key = workshop_id (或全厂 for plant_manager)
  │
  └── 任意 classification_level + 故障诊断/知识召回场景
        → 并行查 Collection 2.2 (长期记忆), partition_key = workshop_id
        → 融合时: 通用文档权重 1.2, 已审核案例权重 1.0, 待验证案例权重 0.7
```

## 三、Elasticsearch 设计

### 3.1 索引定义

```
索引名: manufacturing_knowledge
分片数: 2, 副本数: 1

字段映射:
  chunk_id:      keyword
  doc_id:        keyword
  doc_type:      keyword
  title:         text (ik_smart 分词)
  content:       text (ik_smart 分词)          ← BM25 主检索字段
  workshop_id:   keyword
  equipment_model: text (ik_smart 分词) + keyword (精确匹配子字段)
  fault_code:    keyword                        ← 报警码精确匹配
  classification: keyword
  source:        keyword                        ← sop / case
  created_at:    date
```

### 3.2 BM25 权重参数

```json
{
  "bm25": {
    "k1": 1.2,     // 词频饱和参数
    "b": 0.75       // 文档长度归一化
  }
}
```

### 3.3 字段权重 (搜索时动态指定)

| 字段 | 权重 | 理由 |
|------|------|------|
| fault_code | 3.0 | 报警码精确匹配最优先 |
| equipment_model | 2.0 | 设备型号匹配次优先 |
| content | 1.0 | 正文内容标准权重 |
| title | 1.5 | 标题比正文更重要 |

## 四、Redis Key 设计规范

### 4.1 命名前缀与数据结构

| 前缀 | 用途 | 数据结构 | 过期策略 | 示例 |
|------|------|---------|---------|------|
| `session:` | 会话短期记忆 | LIST (LPUSH/LTRIM) | 会话结束+2h | `session:{session_id}` |
| `checkpoint:` | LangGraph状态 | HASH | 自动续期/主动清理 | `checkpoint:{session_id}:{step}` |
| `idempotent:mes:` | MES回调幂等 | STRING (存"1") | 24h | `idempotent:mes:{mes_event_id}` |
| `idempotent:case:` | 知识沉淀去重 | STRING (存case_id) | 24h | `case:draft:{mes_event_id}` |
| `cache:semantic:` | 语义缓存 | STRING (存response_json) | 文档更新时失效 | `cache:semantic:{hash}:{role}:{workshop}:{classification}` |
| `ratelimit:` | 限流计数器 | STRING (INCR) | 滑动窗口 | `ratelimit:{user_id}:{minute}` |
| `lock:case:` | 知识沉淀分布式锁 | STRING (SET NX) | 30s | `lock:case:{device_id}:{fault_code}` |
| `circuit:` | 熔断器状态 | HASH | 无过期(手动清除) | `circuit:{model_name}` |
| `a2a:sem:` | A2A并发信号量 | STRING (计数器) | 无过期 | `a2a:semaphore` |
| `stream:alarm:` | 告警消息队列 | STREAM | 按消费者组ACK | `stream:alarm:p0` / `stream:alarm:p1` |
| `stream:conversation:` | 对话消息队列 | STREAM | 按消费者组ACK | `stream:conversation:{workshop_id}` |

### 4.2 关键 Key 设计细节

```
# 会话记忆 (最近20轮)
Key:  session:{session_id}
Type: LIST
OP:   LPUSH session:{session_id} {message_json}
      LTRIM session:{session_id} 0 19          # 保留最近20条
TTL:  7200s (2小时, 活跃时自动续期)

# Checkpoint 持久化
Key:  checkpoint:{session_id}:{step_id}
Type: HASH
Fields:
  state_json:   序列化的AgentState
  step_name:    当前Node名
  next_nodes:   待执行Node列表(JSON)
  created_at:   时间戳
  ttl:          3600 (工单未完成时自动续期)
TTL: 3600s (1小时, 关联工单未完成自动续期)

# 语义缓存 (权限隔离)
Key:  cache:semantic:{sha256(query)}_{role}_{workshop_id}_{classification}
Type: STRING
Value: { "response": "...", "citations": [...], "created_at": "..." }
TTL: 3600s (文档更新时主动删除关联缓存)
```

### 4.3 Redis Streams 消费者组设计

```
告警队列:
  stream:alarm:p0  → consumer_group: alarm_processors  (4 workers)
  stream:alarm:p1  → consumer_group: alarm_processors  (2 workers)
  stream:alarm:p2  → consumer_group: alarm_processors  (1 worker)

对话队列:
  stream:conversation:{workshop_id}
    → consumer_group: conv_processors  (2 workers per workshop)

ACK 机制: 消费者处理完消息后 XACK, 未ACK的消息超时30s后 PEL 重试
```
